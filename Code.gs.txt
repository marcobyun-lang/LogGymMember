/************************
 * PT 회원관리 시스템 
 * Backend (Google Apps Script)
 ************************/

const MEMBERS = "Members";
const ATTEND = "Attendance";
const ADMINS = "Admins"; // 관리자 시트

function doGet(e) {
  const action = e.parameter.action;
  
  // 로그인 처리 (GET 방식 대응)
  if (action === "login") {
    return loginCheck(e.parameter.id, e.parameter.password); // password로 수정
  }
  
  if (action === "members") return getMembers();
  if (action === "calendar" || action === "attendance") return getAttendance(e.parameter.memberId);
  if (action === "checkin") return checkIn(e.parameter.memberId);

  return output({ result: "error", message: "Invalid action" });
}

function doPost(e) {
  let data;
  try { data = JSON.parse(e.postData.contents); } catch(ex) { return output({ result: "error" }); }

  // 로그인 처리 (POST 방식 대응)
  if (data.action === "login") {
    return loginCheck(data.id, data.password); // password로 수정
  }
  if (data.action === "checkin") return checkIn(data.memberId);
  
  return output({ result: "error" });
}

/** Admins 시트 기반 로그인 검증 */
function loginCheck(id, pw) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(ADMINS);
  if (!sheet) return output({ result: "fail", message: "Admins 시트 없음" });

  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    // 공백 제거 후 비교
    if (String(rows[i][0]).trim() === String(id).trim() && 
        String(rows[i][1]).trim() === String(pw).trim()) {
      return output({ result: "success", admin: "1" });
    }
  }
  return output({ result: "fail", message: "아이디 또는 비밀번호가 틀립니다." });
}

/** 출석 체크 (시간 오차 해결 적용) */
function checkIn(memberId) {
  const ss = SpreadsheetApp.getActive();
  const mSheet = ss.getSheetByName(MEMBERS);
  const aSheet = ss.getSheetByName(ATTEND);
  const rows = mSheet.getDataRange().getValues();

  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(memberId)) {
      if (rows[i][3] <= 0) return output({ result: "fail", message: "잔여 PT 없음" });

      const now = new Date();
      const today = Utilities.formatDate(now, "Asia/Seoul", "yyyy-MM-dd");
      const nowTime = "'" + Utilities.formatDate(now, "Asia/Seoul", "HH:mm"); // 텍스트 강제 저장

      aSheet.appendRow([memberId, today, nowTime, 1]);
      mSheet.getRange(i + 1, 4).setValue(rows[i][3] - 1);
      return output({ result: "success" });
    }
  }
  return output({ result: "fail", message: "회원 없음" });
}

/** 출석 이력 가져오기 */
function getAttendance(memberId) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(ATTEND);
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  const data = rows.filter(r => String(r[0]) === String(memberId)).map(r => {
    let dateVal = r[1] instanceof Date ? Utilities.formatDate(r[1], "Asia/Seoul", "yyyy-MM-dd") : r[1];
    return { date: dateVal, time: String(r[2]) }; // 시간 텍스트 반환
  });
  return output({ result: "success", data: data });
}

/** 회원 목록 */
function getMembers() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(MEMBERS);
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  const members = rows.filter(r => r[4] === "active").map(r => ({
    id: r[0], name: r[1], phone: r[2], pt: r[3]
  }));
  return output({ result: "success", data: members });
}

function output(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
