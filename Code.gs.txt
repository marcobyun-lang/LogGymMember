/************************
 * PT 회원관리 시스템 Backend (통합 수정본)
 ************************/

const MEMBERS = "Members";
const ATTEND = "Attendance";
const ADMINS = "Admins";

function doGet(e) {
  const action = e.parameter.action;
  if (action === "members") return getMembers();
  if (action === "attendance") return getAttendance(e.parameter.memberId);
  return output({ result: "error" });
}

function doPost(e) {
  let data;
  try { data = JSON.parse(e.postData.contents); } catch(ex) { return output({ result: "error" }); }

  if (data.action === "login") return loginCheck(data.id, data.password);
  if (data.action === "addMember") return addMember(data);
  if (data.action === "updateDetails") return updateMemberDetails(data);
  if (data.action === "deleteMember") return deleteMember(data.memberId);
  if (data.action === "checkin") return checkIn(data.memberId);
  
  return output({ result: "error" });
}

// 1. 회원 등록 (add.html 대응)
function addMember(data) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(MEMBERS);
  // 시트 열 순서: ID, 이름, 강사, PT횟수, 연락처, 주소, 키, 몸무게, 목표, 메모
  const newId = new Date().getTime().toString();
  sheet.appendRow([
    newId, 
    data.name, 
    data.teacher, 
    data.pt, 
    data.phone, 
    data.address, 
    data.height || "", 
    data.weight || "", 
    data.goal, 
    data.memo
  ]);
  return output({ result: "success" });
}

// 2. 회원 정보 수정 (member.html 대응)
function updateMemberDetails(data) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(MEMBERS);
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(data.memberId)) {
      const row = i + 1;
      sheet.getRange(row, 2).setValue(data.name);    // 이름 수정 추가
      sheet.getRange(row, 3).setValue(data.teacher); 
      sheet.getRange(row, 4).setValue(data.pt);      
      sheet.getRange(row, 5).setValue(data.phone);   
      sheet.getRange(row, 6).setValue(data.address); 
      sheet.getRange(row, 7).setValue(data.height);  
      sheet.getRange(row, 8).setValue(data.weight);  
      sheet.getRange(row, 9).setValue(data.goal);    
      sheet.getRange(row, 10).setValue(data.memo);   
      return output({ result: "success" });
    }
  }
  return output({ result: "fail" });
}

// 3. 출석 조회 (한국 시간 포맷)
function getAttendance(memberId) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(ATTEND);
  if (!sheet) return output({ result: "success", data: [] });
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  const data = rows.filter(r => String(r[0]) === String(memberId)).map(r => ({
    date: r[1] instanceof Date ? Utilities.formatDate(r[1], "Asia/Seoul", "yyyy년 MM월 dd일") : String(r[1]),
    time: String(r[2]) // 이미 HH:mm 형식으로 저장된 경우
  }));
  return output({ result: "success", data: data });
}

// 4. 출석 체크 (한국 시간 저장)
function checkIn(memberId) {
  const ss = SpreadsheetApp.getActive();
  const mSheet = ss.getSheetByName(MEMBERS);
  const aSheet = ss.getSheetByName(ATTEND);
  const mRows = mSheet.getDataRange().getValues();
  for (let i = 1; i < mRows.length; i++) {
    if (String(mRows[i][0]) === String(memberId)) {
      if (Number(mRows[i][3]) <= 0) return output({ result: "fail", message: "잔여 횟수가 없습니다." });
      mSheet.getRange(i + 1, 4).setValue(Number(mRows[i][3]) - 1);
      
      const now = new Date();
      const kstDate = Utilities.formatDate(now, "Asia/Seoul", "yyyy-MM-dd");
      const kstTime = Utilities.formatDate(now, "Asia/Seoul", "HH:mm");
      aSheet.appendRow([memberId, kstDate, kstTime]);
      
      return output({ result: "success" });
    }
  }
  return output({ result: "fail" });
}

function getMembers() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(MEMBERS);
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  const members = rows.map(r => ({
    id: r[0], name: r[1], teacher: r[2], pt: r[3], phone: r[4], 
    address: r[5], height: r[6], weight: r[7], goal: r[8], memo: r[9]
  }));
  return output({ result: "success", data: members });
}

function loginCheck(id, pw) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(ADMINS);
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(id).trim() && String(rows[i][1]).trim() === String(pw).trim()) {
      return output({ result: "success", admin: "1" });
    }
  }
  return output({ result: "fail" });
}

function deleteMember(memberId) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(MEMBERS);
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(memberId)) {
      sheet.deleteRow(i + 1);
      return output({ result: "success" });
    }
  }
  return output({ result: "fail" });
}

function output(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
